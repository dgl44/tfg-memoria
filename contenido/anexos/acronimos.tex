\newglossaryentry{algoritmo}{
  name={algoritmo},
  description={Conjunto ordenado y finito de operaciones que permite hallar la solución de un problema. Los algoritmos son la base de la programación y definen los pasos lógicos para resolver tareas computacionales}
}

\newglossaryentry{framework}{
  name={framework},
  plural={frameworks},
  description={Estructura tecnológica de soporte, normalmente con módulos de software concretos, que sirve de base para la organización y el desarrollo de una aplicación}
}

\newglossaryentry{backend}{
  name={backend},
  plural={backends},
  description={Componente que implementa la lógica de un servicio por debajo de la interfaz. En este trabajo, cada backend del analizador da soporte a un lenguaje de programación concreto}
}

\newglossaryentry{pipeline}{
  name={pipeline},
  plural={pipelines},
  description={Secuencia de etapas de procesamiento encadenadas, en la que la salida de cada etapa alimenta la siguiente}
}
